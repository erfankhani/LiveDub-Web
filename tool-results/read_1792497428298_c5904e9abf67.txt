    97→    97→  const CHUNK = 0x8000;
    98→    98→  for (let i = 0; i < bytes.length; i += CHUNK) {

   299→   299→  private handleAudioChunk(chunk: Float32Array) {
   300→   300→    // سطح صدای ورودی (تا ۱۰ بار در ثانیه push شود)
   301→   301→    let max = 0;
   302→   302→    for (let i = 0; i < chunk.length; i++) {
   303→   303→      const a = Math.abs(chunk[i]);
   304→   304→      if (a > max) max = a;
   305→   305→    }
   306→   306→    const now = Date.now();
   307→   307→    if (now - this.lastLevelPush > 100) {
   308→   308→      this.lastLevelPush = now;
   309→   309→      this.cbs.onStats({ inputLevel: Math.min(1, max * 2.2) });
   310→   310→    }
   311→   311→
   312→   312→    const sock = this.activeSocket;
   313→   313→    if (sock && sock.readyState === WebSocket.OPEN) {
   314→   314→      this.sendPcmChunk(sock, chunk);
   315→   315→    } else {
   316→   316→      if (this.pendingChunks.length >= MAX_PENDING_CHUNKS) this.pendingChunks.shift();
   317→   317→      this.pendingChunks.push(chunk);
   318→   318→    }
   319→   319→  }
   320→   320→
   321→   321→  private flushPendingChunks() {
   322→   322→    const sock = this.activeSocket;
   323→   323→    if (!sock || sock.readyState !== WebSocket.OPEN) return;
   324→   324→    while (this.pendingChunks.length) {
   325→   325→      const c = this.pendingChunks.shift();
   326→   326→      if (c) this.sendPcmChunk(sock, c);
   327→   327→    }
   328→   328→  }
   329→   329→
   330→   330→  private sendPcmChunk(sock: WebSocket, f32: Float32Array) {
   331→   331→    if (sock.readyState !== WebSocket.OPEN) return;
   332→   332→    const int16 = new Int16Array(f32.length);
   333→   333→    for (let i = 0; i < f32.length; i++) {
   334→   334→      const s = Math.max(-1, Math.min(1, f32[i]));
   335→   335→      int16[i] = s < 0 ? s * 0x8000 : s * 0x7fff;
   336→   336→    }
   337→   337→    const b64 = bytesToBase64(new Uint8Array(int16.buffer));
   338→   338→    sock.send(
   339→   339→      JSON.stringify({
   340→   340→        realtimeInput: {
   341→   341→          audio: { data: b64, mimeType: "audio/pcm;rate=16000" },
   342→   342→        },
   343→   343→      })
   344→   344→    );
   345→   345→    this.sentBytes += int16.length * 2;
   346→   346→  }

   349→   349→  private openSocket(mode: "initial" | "rotate" | "reconnect"): Promise<void> {
   350→   350→    return new Promise<void>((resolve, reject) => {
   351→   351→      if (!this.running) return reject(new Error("stopped"));
   352→   352→
   353→   353→      const ws = new WebSocket(`${WS_BASE}?key=${encodeURIComponent(this.apiKey)}`);
   354→   354→      let settled = false;
   355→   355→      this.connectingSocket = ws;
   356→   356→
   357→   357→      const connectTimeout = setTimeout(() => {
   358→   358→        if (!settled) {
   359→   359→          try {
   360→   360→            ws.close();
   361→   361→          } catch {
   362→   362→            /* ignore */
   363→   363→          }
   364→   364→        }
   365→   365→      }, CONNECT_TIMEOUT_MS);
   366→   366→
   367→   367→      ws.onopen = () => {
   368→   368→        this.cbs.onLog("info", `WebSocket وصل شد (${mode})`);
   369→   369→        ws.send(
   370→   370→          JSON.stringify({
   371→   371→            setup: {
   372→   372→              model: MODEL,
   373→   373→              generationConfig: {
   374→   374→                responseModalities: ["AUDIO"],
   375→   375→                translationConfig: {
   376→   376→                  targetLanguageCode: this.targetLang,
   377→   377→                  echoTargetLanguage: true,
   378→   378→                },
   379→   379→              },
   380→   380→              inputAudioTranscription: {},
   381→   381→              outputAudioTranscription: {},
   382→   382→            },
   383→   383→          })
   384→   384→        );
   385→   385→      };
   386→   386→
   387→   387→      ws.onmessage = (event) => {
   388→   388→        this.handleServerMessage(ws, event.data, mode).then((ready) => {
   389→   389→          if (ready && !settled) {
   390→   390→            settled = true;
   391→   391→            clearTimeout(connectTimeout);
   392→   392→            this.activateSocket(ws, mode);
   393→   393→            resolve();
   394→   394→          }
   395→   395→        });
   396→   396→      };
   397→   397→
   398→   398→      ws.onerror = () => {
   399→   399→        /* جزئیات در onclose می‌آید */
   400→   400→      };
   401→   401→
   402→   402→      ws.onclose = (e) => {
   403→   403→        clearTimeout(connectTimeout);
   404→   404→        this.cbs.onLog(
   405→   405→          e.code === 1000 ? "info" : "warn",
   406→   406→          `WebSocket بسته شد (کد ${e.code})${e.reason ? ` — ${e.reason}` : ""}`
   407→   407→        );
   408→   408→        if (this.connectingSocket === ws) this.connectingSocket = null;
   409→   409→        if (!settled) {
   410→   410→          settled = true;
   411→   411→          reject(new Error(`اتصال برقرار نشد (کد ${e.code})`));
   412→   412→          return;
   413→   413→        }
   414→   414→        if (ws === this.activeSocket && this.running) {
   415→   415→          this.activeSocket = null;
   416→   416→          this.scheduleReconnect();
   417→   417→        }
   418→   418→      };
   419→   419→    });
   420→   420→  }

   422→   422→  private activateSocket(ws: WebSocket, mode: "initial" | "rotate" | "reconnect") {
   423→   423→    const old = this.activeSocket;
   424→   424→    this.activeSocket = ws;
   425→   425→    this.connectingSocket = null;
   426→   426→
   427→   427→    if (old && old !== ws) {
   428→   428→      try {
   429→   429→        old.onmessage = null;
   430→   430→        old.onclose = null;
   431→   431→        old.onerror = null;
   432→   432→        old.close();
   433→   433→      } catch {
   434→   434→        /* ignore */
   435→   435→      }
   436→   436→    }
   437→   437→
   438→   438→    this.reconnectAttempts = 0;
   439→   439→    this.flushPendingChunks();
   440→   440→
   441→   441→    if (mode === "rotate") {
   442→   442→      this.rotations++;
   443→   443→      this.cbs.onLog("ok", `سشن تمدید شد (#${this.rotations}) — بدون هیچ وقفه‌ای`);
   444→   444→      this.cbs.onStats({ phase: "live", message: `سشن تمدید شد (#${this.rotations}) ✅`, rotations: this.rotations });
   445→   445→    } else if (mode === "reconnect") {
   446→   446→      this.cbs.onLog("ok", "اتصال برقرار شد ✓");
   447→   447→      this.cbs.onStats({ phase: "live", message: "اتصال برقرار شد ✅" });
   448→   448→    }
   449→   449→    this.pushKb();
   450→   450→    this.scheduleRotation();
   451→   451→  }
   452→   452→
   453→   453→  private scheduleRotation() {
   454→   454→    clearTimeout(this.rotationTimer);
   455→   455→    if (!this.running) return;
   456→   456→    this.rotationTimer = setTimeout(() => {
   457→   457→      if (!this.running) return;
   458→   458→      if (this.activeSocket && this.activeSocket.readyState === WebSocket.OPEN && !this.connectingSocket) {
   459→   459→        this.cbs.onLog("info", "Session Rotation خودکار — باز کردن سشن جدید…");
   460→   460→        this.cbs.onStats({ phase: "rotating", message: "تمدید خودکار سشن…" });
   461→   461→        this.openSocket("rotate").catch(() => this.scheduleRotationRetry());
   462→   462→      } else {
   463→   463→        this.scheduleRotation();
   464→   464→      }
   465→   465→    }, SESSION_ROTATE_MS);
   466→   466→  }
   467→   467→
   468→   468→  private scheduleRotationRetry() {
   469→   469→    clearTimeout(this.rotationTimer);
   470→   470→    if (!this.running) return;
   471→   471→    this.rotationTimer = setTimeout(() => {
   472→   472→      if (this.running) this.scheduleRotation();
   473→   473→    }, 5000);
   474→   474→  }
   475→   475→
   476→   476→  private scheduleReconnect() {
   477→   477→    if (!this.running) return;
   478→   478→    clearTimeout(this.reconnectTimer);
   479→   479→    const delay = Math.min(RECONNECT_BASE_MS * Math.pow(2, this.reconnectAttempts), RECONNECT_MAX_MS);
   480→   480→    this.reconnectAttempts++;
   481→   481→    this.reconnects++;
   482→   482→    this.cbs.onLog("warn", `قطع شد؛ تلاش مجدد در ${(delay / 1000).toFixed(1)}s (تلاش ${this.reconnectAttempts})`);
   483→   483→    this.cbs.onStats({
   484→   484→      phase: "reconnecting",
   485→   485→      message: `قطع شد؛ اتصال مجدد در ${(delay / 1000).toFixed(1)} ثانیه…`,
   486→   486→      reconnects: this.reconnects,
   487→   487→    });
   488→   488→    this.pushKb();
   489→   489→    this.reconnectTimer = setTimeout(() => {
   490→   490→      if (!this.running) return;
   491→   491→      this.openSocket("reconnect").catch(() => this.scheduleReconnect());
   492→   492→    }, delay);
   493→   493→  }

   495→   495→  // ── پیام‌های سرور ──
   496→   496→  private async handleServerMessage(
   497→   497→    ws: WebSocket,
   498→   498→    data: unknown,
   499→   499→    mode: "initial" | "rotate" | "reconnect"
   500→   500→  ): Promise<boolean> {
   501→   501→    let text: string;
   502→   502→    if (data instanceof Blob) {
   503→   503→      text = await data.text();
   504→   504→    } else if (typeof data === "string") {
   505→   505→      text = data;
   506→   506→    } else {
   507→   507→      return false;
   508→   508→    }
   509→   509→
   510→   510→    let msg: Record<string, unknown>;
   511→   511→    try {
   512→   512→      msg = JSON.parse(text) as Record<string, unknown>;
   513→   513→    } catch {
   514→   514→      return false;
   515→   515→    }
   516→   516→
   517→   517→    if (msg.setupComplete) {
   518→   518→      this.cbs.onLog("ok", `Setup complete (${mode}) ✓`);
   519→   519→      return true;
   520→   520→    }
   521→   521→
   522→   522→    // ⭐ خطاهای سرور — مثل کلید نامعتبر / دسترسی به مدل / کوتا
   523→   523→    if (msg.error) {
   524→   524→      const err = msg.error as { message?: string; code?: number; status?: string };
   525→   525→      const m = `[${err.code ?? "?"} ${err.status ?? ""}] ${err.message ?? "خطای ناشناخته سرور"}`;
   526→   526→      this.serverError = m;
   527→   527→      this.cbs.onLog("error", `خطای سرور: ${m}`);
   528→   528→      this.cbs.onStats({ serverError: m, message: "خطای سرور Gemini — جزئیات در لاگ" });
   529→   529→      return false;
   530→   530→    }
   531→   531→
   532→   532→    if (msg.goAway) {
   533→   533→      this.cbs.onLog("info", "سرور goAway فرستاد → تمدید فوری سشن");
   534→   534→      if (this.running && this.activeSocket === ws && !this.connectingSocket) {
   535→   535→        this.openSocket("rotate").catch(() => this.scheduleRotationRetry());
   536→   536→      }
   537→   537→      return false;
   538→   538→    }
   539→   539→
   540→   540→    const content = msg.serverContent as
   541→   541→      | {
   542→   542→          interrupted?: boolean;
   543→   543→          inputTranscription?: { text?: string };
   544→   544→          outputTranscription?: { text?: string };
   545→   545→          modelTurn?: { parts?: Array<{ inlineData?: { data?: string } }> };
   546→   546→        }
   547→   547→      | undefined;
   548→   548→    if (!content) return false;
   549→   549→
   550→   550→    if (content.interrupted && this.playbackCtx) {
   551→   551→      this.nextPlayTime = this.playbackCtx.currentTime + 0.005;
   552→   552→    }
   553→   553→
   554→   554→    if (content.inputTranscription?.text) {
   555→   555→      this.cbs.onTranscript("in", content.inputTranscription.text);
   556→   556→    }
   557→   557→    if (content.outputTranscription?.text) {
   558→   558→      this.cbs.onTranscript("out", content.outputTranscription.text);
   559→   559→    }
   560→   560→
   561→   561→    if (content.modelTurn?.parts) {
   562→   562→      for (const part of content.modelTurn.parts) {
   563→   563→        if (part.inlineData?.data) {
   564→   564→          this.recvAudioBytes += (part.inlineData.data.length * 3) / 4;
   565→   565→          this.playTranslatedAudio(part.inlineData.data);
   566→   566→        }
   567→   567→      }
   568→   568→      this.pushKb();
   569→   569→    }
   570→   570→    return false;
   571→   571→  }
   572→   572→
   573→   573→  // ── پخش صدای دوبله ──
   574→   574→  private playTranslatedAudio(b64: string) {
   575→   575→    if (!this.playbackCtx || !this.masterGain) return;
   576→   576→    try {
   577→   577→      const binary = atob(b64);
   578→   578→      const bytes = new Uint8Array(binary.length);
   579→   579→      for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
   580→   580→      const int16 = new Int16Array(bytes.buffer);
   581→   581→      const f32 = new Float32Array(int16.length);
   582→   582→      for (let i = 0; i < int16.length; i++) f32[i] = int16[i] / 32768;
   583→   583→
   584→   584→      const buffer = this.playbackCtx.createBuffer(1, f32.length, 24000);
   585→   585→      buffer.getChannelData(0).set(f32);
   586→   586→      const src = this.playbackCtx.createBufferSource();
   587→   587→      src.buffer = buffer;
   588→   588→      src.connect(this.masterGain);
   589→   589→
   590→   590→      const now = this.playbackCtx.currentTime;
   591→   591→      if (this.nextPlayTime < now || this.nextPlayTime - now > 0.08) {
   592→   592→        this.nextPlayTime = now + 0.005;
   593→   593→      }
   594→   594→      src.start(this.nextPlayTime);
   595→   595→      this.nextPlayTime += buffer.duration;
   596→   596→    } catch (e) {
   597→   597→      this.cbs.onLog("error", `خطای پخش: ${e instanceof Error ? e.message : String(e)}`);
   598→   598→    }
   599→   599→  }
   600→   600→
   601→   601→  private startOutputMeter() {
   602→   602→    const analyser = this.analyser;
   603→   603→    if (!analyser) return;
   604→   604→    const buf = new Float32Array(analyser.fftSize);
   605→   605→    this.analyserTimer = setInterval(() => {
   606→   606→      if (!this.analyser) return;
   607→   607→      this.analyser.getFloatTimeDomainData(buf);
   608→   608→      let peak = 0;
   609→   609→      for (let i = 0; i < buf.length; i++) {
   610→   610→        const a = Math.abs(buf[i]);
   611→   611→        if (a > peak) peak = a;
   612→   612→      }
   613→   613→      this.cbs.onStats({ outputLevel: Math.min(1, peak * 2.2) });
   614→   614→    }, 120);
   615→   615→  }
   616→   616→
   617→   617→  private pushKb() {
   618→   618→    this.cbs.onStats({
   619→   619→      sentKb: Math.round(this.sentBytes / 1024),
   620→   620→      recvKb: Math.round(this.recvAudioBytes / 1024),
   621→   621→    });
   622→   622→  }
   623→   623→
   624→   624→  get errorText(): string {
   625→   625→    return this.serverError;
   626→   626→  }
   627→   627→}
